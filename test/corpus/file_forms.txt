PROGRAM _INIT
    nCounter := 0;
    state := PROBE_IDLE;
    cmd.nSetpoint := 100;
    pCounter := ADR(nCounter);
END_PROGRAM

PROGRAM _CYCLIC
    rtrigStart(CLK := cmd.bStart);
    IF bEnable AND (nCounter < 1000) THEN
        nCounter := nCounter + nStep;
    END_IF;
    CASE state OF
        PROBE_IDLE:
            IF rtrigStart.Q THEN
                state := PROBE_STARTING;
            END_IF;
        PROBE_RUNNING:
            fbMotor(bEnable := TRUE, nSpeed := 10);
        PROBE_STOPPING, PROBE_FAULT:
            fbMotor(bEnable := FALSE, nSpeed := 0);
    END_CASE;
    nCounter := ProbeAdd(nA := nCounter, nB := 1);
END_PROGRAM

PROGRAM _EXIT
    nCounter := 0;
END_PROGRAM
